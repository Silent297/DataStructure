```cpp
#include <vector>

bool Is_legal(std::vector<char> &opt){
    std::vector<int> stack;
    for(size_t i = 0; i < opt.size(); i++){
        if(opt[i] == 'I')
            stack.push_back(1);
        else if(opt[i] == 'O'){
            if(stack.empty())
                return false;
            else{
                stack.pop_back();
            }
        }
    }
    if(!stack.empty())
        return false;
    else
        return true;
}
```

#### 第 2 问

```cpp
#include <stack>

enum class State{
    Waiting,Calculate
};

typedef struct ack{
    int m;
    int n;
    State state;      //用于指示
}ack;


int Ack2(int m,int n){
    std::stack<ack> s;
    s.push({m, n, State::Calculate});

    while(s.empty() == false){
        ack cur = s.top();
        s.pop();

        if(cur.state == State::Calculate){
            if(cur.m == 0){
                int result = cur.n + 1;
                if(s.empty()){
                    return result;
                }
                else{
                    ack last = s.top();
                    s.pop();
                    last.n = result;
                    last.state = State::Calculate;
                    s.push(last);
                }
            }
            else if(cur.n == 0){
                s.push({cur.m-1, 1, State::Calculate});
            }
            else{
                s.push({cur.m-1, 0, State::Waiting});   //0表示n空缺，等待内层计算结果
                s.push({cur.m, cur.n-1, State::Calculate});
            }
        }
        else{
            s.push({cur.m, cur.n, State::Calculate});   //应该不会到达这个分支，前面的选择已经处理过了
        }
    }
    return 0;  //这里正常也是不会到达的
}
```

### 第 10 题

```cpp
#include "LNode.h"

int Max_int(LinkList f){
    if(f == NULL)
        return INT_MIN;
    int rest_max = Max_int(f->next);
    if(f->data >= rest_max)
        return f->data;
    else
        return rest_max;
}

int Node_amount(LinkList f){
    if(f == NULL)
        return 0;
    else
        return Node_amount(f->next) + 1;
}

int Sum(LinkList f){
    if(f == NULL)
        return 0;
    else
        return f->data + Sum(f->next);
}

double Mean(LinkList f){
    return (double)Sum(f)/Node_amount(f);
}
```

```cpp
#include <iostream>
#include <vector>
#include <string>

int main() {
    const int N = 8;              
    const int TOTAL = 6561;       
    char choices[3] = {' ', '+', '-'};
   for (int opt_num = 0; opt_num < TOTAL; opt_num++) {
        int temp = opt_num;
        std::vector<char> opts(N);
        for (int i = N - 1; i >= 0; i--) {
            opts[i] = choices[temp % 3];
            temp /= 3;
        }

        std::string expr = "1";
        int result = 0;
        int cur_num = 1;
        char sign = '+';

        for (int i = 0; i < N; i++) {
            int digit = i + 2;
            char op = opts[i];

            if (op == ' ') {
                cur_num = cur_num * 10 + digit;
                expr += std::to_string(digit);
            } else {
                if (sign == '+') 
                    result += cur_num;
                else             
                    result -= cur_num;

                sign = op;
                cur_num = digit;
                expr += op;
                expr += std::to_string(digit);
            }
        }

        if (sign == '+')   
            result += cur_num;
        else            
            result -= cur_num;

        if (result == 110) {
            std::cout << expr << " = 110" << std::endl;
        }
    }

    return 0;
}
```
